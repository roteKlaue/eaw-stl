# STL_neighbors_help: the mountain road to Stalliongrad reopens.
country_event = {
    id = stl_equality.1
    title = stl_equality.1.t
    desc = stl_equality.1.d
    picture = GFX_event_stalliongrad

    is_triggered_only = yes
    fire_only_once = yes

    option = {
        name = stl_equality.1.a
        add_political_power = 50
        if = {
            limit = { country_exists = STG }
            add_opinion_modifier = { target = STG modifier = worker_solidarity2 }
            STG = {
                add_opinion_modifier = { target = ROOT modifier = worker_solidarity2 }
            }
        }
        swap_ideas = { remove_idea = STL_city_state add_idea = STL_city_state2 }
    }
}
