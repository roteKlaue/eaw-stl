STL_can_increase_infiltration_in_state = {
    OR = {
        NOT = { has_state_flag = STL_infiltration_@PREV }
        has_state_flag = {
            flag = STL_infiltration_@PREV
            value < 3
        }
    }
}

STL_is_infiltrated_by_ROOT = {
	has_state_flag = {
		flag = STL_infiltration_@PREV
		value > 0
	}
}

STL_economy_cd_over = {
	NOT = {
		has_country_flag = {
			flag = STL_economy_cd_flag
			days < 240
		}
	}
}

STL_economy_check_1 = {

		has_country_flag = formed_stg_stl
		is_in_faction_with = EQS
		has_completed_focus = STL_economic_independence
	}
}
